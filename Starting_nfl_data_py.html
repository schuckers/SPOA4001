The next set of code aggregates all of the play by play data by team. We get some variables based upon expected points added (EPA) per play. We do this aggregation by offense and defense then join them back together.

Show Python Code 
# Offensive Aggregation (posteam)
offense_summary = pbp_w1_4.groupby('posteam').apply(
    lambda x: pd.Series({
        'off_plays': len(x),
        'off_epa_per_play': x['epa'].mean(),
        'pass_epa': x.loc[x['pass'] == 1, 'epa'].mean(),
        'rush_epa': x.loc[x['rush'] == 1, 'epa'].mean(),
        'off_success_rate': x['success'].mean(),
        'dropback_pct': (x['pass'] == 1).mean()
    }),
    include_groups=False
).reset_index()

# Defensive Aggregation (defteam)
defense_summary = pbp_w1_4.groupby('defteam').apply(
    lambda x: pd.Series({
        'def_plays': len(x),
        'def_epa_allowed': x['epa'].mean(),
        'def_pass_epa_allowed': x.loc[x['pass'] == 1, 'epa'].mean(),
        'def_rush_epa_allowed': x.loc[x['rush'] == 1, 'epa'].mean(),
        'def_success_rate_allowed': x['success'].mean()
    }),
    include_groups=False
).reset_index()

# Join Offense & Defense into a single Master Team Summary

# Join metrics and classify into performance tiers
all_ypp = pd.merge(off_ypp, def_ypp, on='team')

mean_off = all_ypp['off_ypp'].mean()
mean_def = all_ypp['def_ypp'].mean()

conditions = [
    (all_ypp['off_ypp'] > mean_off) & (all_ypp['def_ypp'] < mean_def),
    (all_ypp['off_ypp'] > mean_off) & (all_ypp['def_ypp'] >= mean_def),
    (all_ypp['off_ypp'] <= mean_off) & (all_ypp['def_ypp'] < mean_def)
]
choices = ["Elite", "High Offense / Low Defense", "Low Offense / High Defense"]
all_ypp['tier'] = np.select(conditions, choices, default="Rebuilding")

all_ypp['net_ypp'] = all_ypp['off_ypp'] - all_ypp['def_ypp']
all_ypp = all_ypp.sort_values(by='net_ypp', ascending=False)
