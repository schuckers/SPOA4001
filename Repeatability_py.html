].copy()

# Weekly Offensive Metrics
weekly_offense = (
    pbp_multi.groupby(['season', 'week', 'posteam'])
    .apply(
        lambda x: pd.Series({
            'off_plays': len(x),
            'off_epa_per_play': x['epa'].mean(),
            'off_pass_epa': x.loc[x['pass'] == 1, 'epa'].mean(),
            'off_rush_epa': x.loc[x['rush'] == 1, 'epa'].mean(),
            'off_success_rate': x['success'].mean(),
            'pass_rate': (x['pass'] == 1).mean(),
            'complete_pass_pct': x['complete_pass'].mean()
        }),
        include_groups=False
    )
    .reset_index()
)

# Weekly Defensive Metrics
weekly_defense = (
    pbp_multi.groupby(['season', 'week', 'defteam'])
    .agg(
        def_plays=('epa', 'count'),
        def_epa_allowed=('epa', 'mean'),
        def_success_rate_allowed=('success', 'mean')
    )
    .reset_index()
)

# Master Panel Dataset (Weekly Team Summary)
weekly_team_panel = (
    pd.merge(

# Get just the 2025 NFL play-by-play data
pbp_raw = pbp_multi[(pbp_multi['season'])==2025]

# Filter to regular season scrimmage plays
pbp_clean = pbp_raw[
    (pbp_raw['season_type'] == 'REG') &
    (pbp_raw['play'] == 1) &
    (pbp_raw['epa'].notna()) &
    (pbp_raw['posteam'].notna()) &
    (pbp_raw['play_type'].isin(['pass', 'run']))
].copy()

# Add indicators
pbp_clean['half_season'] = np.where(pbp_clean['week'] <= 9, "H1", "H2")
pbp_clean['is_explosive'] = np.where(pbp_clean['yards_gained'] >= 15, 1, 0)
pbp_clean['is_success'] = np.where(pbp_clean['epa'] > 0, 1, 0)
pbp_clean['turnover'] = ((pbp_clean['interception'] == 1) | (pbp_clean['fumble_lost'] == 1)).astype(int)

Turn the play-by-play data into drive-level data:

# Drive-Level Metrics (drive_epa, turnovers_per_drive)
drive_game = (
    pbp_clean.groupby(['game_id', 'drive', 'posteam', 'half_season'])
    .agg(
        drive_total_epa=('epa', 'sum'),
        drive_turnovers=('turnover', 'sum')
    )
    .reset_index()
)

drive_summary = (
    drive_game.groupby(['posteam', 'half_season'])
    .agg(
        drive_epa=('drive_total_epa', 'mean'),
        turnovers_per_drive=('drive_turnovers', 'mean')
    )
    .reset_index()
)

# Play-Level Metrics
play_summary = (
    pbp_clean.groupby(['posteam', 'half_season'])
    .apply(
        lambda x: pd.Series({
            'mean_epa': x['epa'].mean(),
            'success_rate': x['is_success'].mean(),
            'pass_epa': x.loc[x['play_type'] == 'pass', 'epa'].mean(),
            'rush_epa': x.loc[x['play_type'] == 'run', 'epa'].mean(),
            'yards_per_play': x['yards_gained'].mean(),
            'completion_pct': x.loc[x['pass_attempt'] == 1, 'complete_pass'].mean()
        }),
        include_groups=False
    )
    .reset_index()
)

# Combine metrics
metrics_combined = pd.merge(play_summary, drive_summary, on=['posteam', 'half_season'])
metrics_combined.head()

Reshape the data to calculate split-half correlations:

# Pivot to wide format: metric_H1 vs metric_H2
metrics_wide = metrics_combined.pivot(index='posteam', columns='half_season')
metrics_wide.columns = [f"{val}_{stat}" for val, stat in metrics_wide.columns]
metrics_wide = metrics_wide.reset_index()

metric_list = [
    "mean_epa", "success_rate", "pass_epa", "rush_epa",
    "yards_per_play", "drive_epa", "completion_pct", "turnovers_per_drive"
]

Next calculate the correlations between the first half and the second half

# Compute Pearson r, p-value, and R-squared
corr_data = []

for m in metric_list:
    col_h1 = f"{m}_H1"
    col_h2 = f"{m}_H2"
    
    valid_data = metrics_wide[[col_h1, col_h2]].dropna()
    r, p_val = stats.pearsonr(valid_data[col_h1], valid_data[col_h2])
    
    corr_data.append({
        'metric': m,
        'r': r,
        'p_value': p_val,
        'r_squared': r ** 2
    })

correlation_results = pd.DataFrame(corr_data).sort_values(by='r', ascending=False)
# Render Summary Table using Great Tables
GT(correlation_results).tab_header(
    title="2025 NFL Offense Split-Half Reliability",
    subtitle="Pearson Correlation (r) between Weeks 1-9 (H1) and Weeks 10-18 (H2)"
).cols_label(
    metric="Offensive Metric",
    r="Pearson r",
    p_value="p-value",
    r_squared="R²"
).fmt_number(
    columns=["r", "p_value", "r_squared"], 
    decimals=3
)
